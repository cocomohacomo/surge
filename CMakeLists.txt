cmake_minimum_required(VERSION 3.16)
project(surge_core VERSION 0.9.11 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(SURGE_ENABLE_SANITIZERS "Build all targets with AddressSanitizer and UBSan" OFF)
option(SURGE_BUILD_FUZZERS "Build libFuzzer targets (Clang only)" OFF)
option(SURGE_BUILD_BENCHMARKS "Build opt-in performance benchmarks" OFF)

if(SURGE_ENABLE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "SURGE_ENABLE_SANITIZERS currently requires GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(surge_core SHARED src/c_api.cpp src/core.cpp src/event_journal.cpp src/context_graph.cpp src/dependency_engine.cpp src/question_router.cpp src/scope_validator.cpp src/context_hint.cpp src/packet_admission.cpp src/sha256.cpp src/temporal_boundary_ledger.cpp src/temporal_question_ledger.cpp src/evidence_provenance.cpp src/chaos_cluster.cpp)
target_include_directories(surge_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(SURGE_BUILD_BENCHMARKS)
  add_executable(surge_journal_bench benchmarks/event_journal_bench.cpp)
  target_link_libraries(surge_journal_bench PRIVATE surge_core)
endif()

if(SURGE_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "SURGE_BUILD_FUZZERS requires Clang/libFuzzer")
  endif()
  target_compile_options(surge_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(surge_core PRIVATE -fsanitize=address,undefined)
  add_executable(surge_journal_fuzz fuzz/journal_replay_fuzz.cpp)
  target_link_libraries(surge_journal_fuzz PRIVATE surge_core)
  target_compile_options(surge_journal_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(surge_journal_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
set_target_properties(surge_core PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(surge_core PRIVATE /W4 /WX)
else()
  target_compile_options(surge_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(surge_chaos_c_api_tests tests/chaos_c_api_tests.cpp)
target_link_libraries(surge_chaos_c_api_tests PRIVATE surge_core)
add_executable(surge_core_tests tests/core_tests.cpp)
target_link_libraries(surge_core_tests PRIVATE surge_core)
add_executable(surge_journal_tests tests/journal_tests.cpp)
target_link_libraries(surge_journal_tests PRIVATE surge_core)
add_executable(surge_chaos_cluster_tests tests/chaos_cluster_tests.cpp)
target_link_libraries(surge_chaos_cluster_tests PRIVATE surge_core)
add_executable(surge_context_graph_tests tests/context_graph_tests.cpp)
target_link_libraries(surge_context_graph_tests PRIVATE surge_core)
add_executable(surge_dependency_engine_tests tests/dependency_engine_tests.cpp)
target_link_libraries(surge_dependency_engine_tests PRIVATE surge_core)
add_executable(surge_question_router_tests tests/question_router_tests.cpp)
target_link_libraries(surge_question_router_tests PRIVATE surge_core)
add_executable(surge_scope_validator_tests tests/scope_validator_tests.cpp)
target_link_libraries(surge_scope_validator_tests PRIVATE surge_core)
add_executable(surge_packet_admission_tests tests/packet_admission_tests.cpp)
target_link_libraries(surge_packet_admission_tests PRIVATE surge_core)
add_executable(surge_context_hint_tests tests/context_hint_tests.cpp)
target_link_libraries(surge_context_hint_tests PRIVATE surge_core)
add_executable(surge_runtime_pipeline_tests tests/runtime_pipeline_tests.cpp)
add_executable(surge_temporal_boundary_ledger_tests tests/temporal_boundary_ledger_tests.cpp)
target_link_libraries(surge_temporal_boundary_ledger_tests PRIVATE surge_core)
add_executable(surge_temporal_question_ledger_tests tests/temporal_question_ledger_tests.cpp)
target_link_libraries(surge_temporal_question_ledger_tests PRIVATE surge_core)
add_executable(surge_atomic_semantic_tests tests/atomic_semantic_transaction_tests.cpp)
add_executable(surge_evidence_provenance_tests tests/evidence_provenance_tests.cpp)
add_executable(surge_classification_proposal_tests tests/classification_proposal_journal_tests.cpp)
target_link_libraries(surge_classification_proposal_tests PRIVATE surge_core)
target_link_libraries(surge_evidence_provenance_tests PRIVATE surge_core)
target_link_libraries(surge_atomic_semantic_tests PRIVATE surge_core)
target_link_libraries(surge_runtime_pipeline_tests PRIVATE surge_core)
